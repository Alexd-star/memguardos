# 持久化 Kylin 向量与 RRF 检索 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将 Kylin SDK 从查询时逐候选推理改为写入后持久化向量，并用可解释的意图路由与 RRF 排名融合稳定提升检索质量、并发尾延迟和精准遗忘一致性。

**Architecture:** SQLite 保存带内容哈希、后端和模型修订号的归一化向量；有界后台回填只处理可信且处于 active 状态的记忆，查询线程最多生成一次查询向量并只读取已持久化的候选向量，缺失时立即保留 hash 检索结果。词法、hash 向量、Kylin 向量和按意图选择的图谱检索各自独立排序，再通过加权 RRF 融合；精确短语和实体强命中受保护，图谱仅在关系型查询中参与。

**Tech Stack:** Python 3.10+、SQLite WAL、`struct` 二进制向量编码、现有 Kylin AI SDK 子进程封装、pytest、ruff、MemGuardOS 消融与并发评测脚本。

```python
# memguard/retrieval/vector_cache.py
from __future__ import annotations

from dataclasses import dataclass
import hashlib
import math
import struct
from typing import Iterable, List


@dataclass(frozen=True)
class PersistedVector:
    memory_id: str
    content_hash: str
    backend: str
    model_revision: str
    dimension: int
    vector: List[float]
    status: str


def vector_content_hash(summary: str, content: str) -> str:
    payload = (str(summary) + "\0" + str(content)).encode("utf-8")
    return hashlib.sha256(payload).hexdigest()


def encode_vector(values: Iterable[float]) -> bytes:
    vector = [float(value) for value in values]
    if not vector:
        raise ValueError("vector dimension must be positive")
    if not all(math.isfinite(value) for value in vector):
        raise ValueError("vector values must be finite")
    return struct.pack("<%sf" % len(vector), *vector)


def decode_vector(blob: bytes, dimension: int) -> List[float]:
    if dimension <= 0 or len(blob) != dimension * 4:
        raise ValueError("vector blob does not match declared dimension")
    return list(struct.unpack("<%sf" % dimension, blob))
```

```python
def upsert_memory_embedding(
    self,
    memory_id: str,
    content_hash: str,
    backend: str,
    model_revision: str,
    vector: List[float],
) -> None:
    from memguard.retrieval.vector_cache import encode_vector

    now = utc_now()
    blob = encode_vector(vector)
    self.conn.execute(
        """
        INSERT INTO memory_embedding(
            memory_id, content_hash, backend, model_revision, dimension,
            vector_blob, status, failure_reason, created_at, updated_at
        ) VALUES (?, ?, ?, ?, ?, ?, 'ready', '', ?, ?)
        ON CONFLICT(memory_id, backend, model_revision, content_hash) DO UPDATE SET
            dimension=excluded.dimension, vector_blob=excluded.vector_blob,
            status='ready', failure_reason='', updated_at=excluded.updated_at
        """,
        (memory_id, content_hash, backend, model_revision, len(vector), blob, now, now),
    )
    self.conn.commit()

def get_memory_embedding(
    self, memory_id: str, content_hash: str, backend: str, model_revision: str
):
    from memguard.retrieval.vector_cache import PersistedVector, decode_vector

    row = self.conn.execute(
        """SELECT * FROM memory_embedding
           WHERE memory_id=? AND content_hash=? AND backend=? AND model_revision=?
             AND status='ready'""",
        (memory_id, content_hash, backend, model_revision),
    ).fetchone()
    if row is None or row["vector_blob"] is None:
        return None
    self.conn.execute(
        """UPDATE memory_embedding SET last_used_at=?
           WHERE memory_id=? AND content_hash=? AND backend=? AND model_revision=?""",
        (utc_now(), memory_id, content_hash, backend, model_revision),
    )
    self.conn.commit()
    return PersistedVector(
        memory_id=row["memory_id"], content_hash=row["content_hash"],
        backend=row["backend"], model_revision=row["model_revision"],
        dimension=int(row["dimension"]),
        vector=decode_vector(row["vector_blob"], int(row["dimension"])),
        status=row["status"],
    )

def delete_memory_embeddings(self, memory_id: str) -> int:
    cursor = self.conn.execute("DELETE FROM memory_embedding WHERE memory_id=?", (memory_id,))
    self.conn.commit()
    return int(cursor.rowcount)

def embedding_cache_status(self, backend: str, model_revision: str) -> Dict[str, Any]:
    rows = self.conn.execute(
        """SELECT status, COUNT(*) AS count FROM memory_embedding
           WHERE backend=? AND model_revision=? GROUP BY status""",
        (backend, model_revision),
    ).fetchall()
    counts = {"pending": 0, "ready": 0, "failed": 0}
    counts.update({str(row["status"]): int(row["count"]) for row in rows})
    return {"backend": backend, "model_revision": model_revision, **counts}
```

```python
# memguard/store/sqlite_meta.py
def enqueue_memory_embedding(
    self, memory_id: str, content_hash: str, backend: str, model_revision: str
) -> None:
    now = utc_now()
    self.conn.execute(
        """
        INSERT INTO memory_embedding(
            memory_id, content_hash, backend, model_revision, dimension,
            vector_blob, status, failure_reason, created_at, updated_at
        ) VALUES (?, ?, ?, ?, 0, NULL, 'pending', '', ?, ?)
        ON CONFLICT(memory_id, backend, model_revision, content_hash) DO UPDATE SET
            status='pending', failure_reason='', updated_at=excluded.updated_at
        """,
        (memory_id, content_hash, backend, model_revision, now, now),
    )
    self.conn.commit()

def list_pending_embeddings(
    self, backend: str, model_revision: str, limit: int
) -> List[Dict[str, Any]]:
    rows = self.conn.execute(
        """
        SELECT e.memory_id, e.content_hash, m.summary, m.content
        FROM memory_embedding e JOIN memory m ON m.memory_id=e.memory_id
        WHERE e.backend=? AND e.model_revision=? AND e.status IN ('pending', 'failed')
          AND m.deleted=0 AND m.governance_state='active'
          AND m.trust_level IN ('trusted', 'verified', 'reviewed')
        ORDER BY CASE e.status WHEN 'pending' THEN 0 ELSE 1 END, e.updated_at
        LIMIT ?
        """,
        (backend, model_revision, max(1, int(limit))),
    ).fetchall()
    return [dict(row) for row in rows]

def mark_embedding_failed(
    self, memory_id: str, content_hash: str, backend: str,
    model_revision: str, reason: str
) -> None:
    self.conn.execute(
        """UPDATE memory_embedding SET status='failed', failure_reason=?, updated_at=?
           WHERE memory_id=? AND content_hash=? AND backend=? AND model_revision=?""",
        (str(reason)[:500], utc_now(), memory_id, content_hash, backend, model_revision),
    )
    self.conn.commit()
```

```python
# memguard/service.py
def embedding_cache_status(self) -> Dict[str, Any]:
    return self.meta.embedding_cache_status(
        self.EMBEDDING_BACKEND, self.EMBEDDING_MODEL_REVISION
    )

def backfill_embeddings(self, limit: int = 100, timeout_seconds: float = 30.0) -> Dict[str, Any]:
    started = time.monotonic()
    deadline = started + max(0.05, float(timeout_seconds))
    provider = self.retriever.kylin_provider
    rows = self.meta.list_pending_embeddings(
        self.EMBEDDING_BACKEND, self.EMBEDDING_MODEL_REVISION, max(1, int(limit))
    )
    ready = 0
    failed = 0
    memory_ids: List[str] = []
    for row in rows:
        remaining = deadline - time.monotonic()
        if remaining <= 0:
            break
        text = "%s\n%s" % (row["summary"], row["content"])
        vector = provider.vector(text, timeout_seconds=remaining) if provider else None
        if vector:
            self.meta.upsert_memory_embedding(
                row["memory_id"], row["content_hash"], self.EMBEDDING_BACKEND,
                self.EMBEDDING_MODEL_REVISION, vector,
            )
            ready += 1
            memory_ids.append(row["memory_id"])
        else:
            self.meta.mark_embedding_failed(
                row["memory_id"], row["content_hash"], self.EMBEDDING_BACKEND,
                self.EMBEDDING_MODEL_REVISION, "Kylin SDK 未在时间预算内返回有效向量",
            )
            failed += 1
    status = self.embedding_cache_status()
    return {
        "processed": ready + failed,
        "ready": ready,
        "failed": failed,
        "remaining": status["pending"],
        "elapsed_ms": round((time.monotonic() - started) * 1000.0, 3),
        "memory_ids": memory_ids,
    }
```

```python
# memguard/retrieval/fusion.py
from __future__ import annotations

from dataclasses import dataclass
from enum import Enum
import re
from typing import Dict, Iterable, List, Mapping, Optional, Set, Tuple


class QueryIntent(str, Enum):
    EXACT_LOOKUP = "exact_lookup"
    PREFERENCE = "preference"
    PROCEDURAL = "procedural"
    TEMPORAL = "temporal"
    RELATIONAL = "relational"
    GENERAL = "general"


@dataclass(frozen=True)
class FusedResult:
    memory_id: str
    score: float
    sources: Tuple[str, ...]
    protected: bool = False


def classify_query_intent(query: str) -> QueryIntent:
    text = str(query or "").lower()
    if re.search(r"\bmem(?:ory)?[-_ ]?id\b|\bmem-[a-z0-9-]+\b", text):
        return QueryIntent.EXACT_LOOKUP
    if any(term in text for term in ("偏好", "习惯", "明确说", "prefer", "preference")):
        return QueryIntent.PREFERENCE
    if any(term in text for term in ("步骤", "怎么修", "失败后", "流程", "procedure", "fix")):
        return QueryIntent.PROCEDURAL
    if any(term in text for term in ("上周", "昨天", "之前", "最新", "更新后", "when", "latest")):
        return QueryIntent.TEMPORAL
    if any(term in text for term in ("关系", "关联", "依赖", "属于", "related", "relationship")):
        return QueryIntent.RELATIONAL
    return QueryIntent.GENERAL


def rrf_fuse(
    rankings: Mapping[str, Iterable[str]],
    weights: Optional[Mapping[str, float]] = None,
    k: int = 60,
    exact_hits: Optional[Set[str]] = None,
) -> List[FusedResult]:
    if k <= 0:
        raise ValueError("RRF k must be positive")
    totals: Dict[str, float] = {}
    sources: Dict[str, List[str]] = {}
    for source, ranking in rankings.items():
        weight = float((weights or {}).get(source, 1.0))
        seen: Set[str] = set()
        for rank, memory_id in enumerate(ranking, start=1):
            if memory_id in seen:
                continue
            seen.add(memory_id)
            totals[memory_id] = totals.get(memory_id, 0.0) + weight / float(k + rank)
            sources.setdefault(memory_id, []).append(source)
    protected = exact_hits or set()
    return sorted(
        (
            FusedResult(mid, score, tuple(sources[mid]), mid in protected)
            for mid, score in totals.items()
        ),
        key=lambda item: (not item.protected, -item.score, item.memory_id),
    )
```

```python
# scripts/evaluate_retrieval_v2.py
from __future__ import annotations

import argparse
import hashlib
import json
from pathlib import Path
import subprocess
import time

from memguard.eval.ablation import QUERY_PATH, SEED_PATH, run_ablation


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for chunk in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def git_head() -> str:
    return subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()


def evaluate_gate(rounds):
    failures = []
    for index, row in enumerate(rounds, start=1):
        if float(row["recall_at_5"]) < 1.0:
            failures.append(f"round {index}: recall_at_5 < 1.0")
        if float(row["p95_latency_ms"]) >= 500.0:
            failures.append(f"round {index}: p95_latency_ms >= 500")
        if float(row["p99_latency_ms"]) >= 500.0:
            failures.append(f"round {index}: p99_latency_ms >= 500")
    return {"pass": not failures, "failures": failures}


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--rounds", type=int, default=3)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    rounds = []
    for index in range(max(1, args.rounds)):
        started = time.time()
        result = run_ablation(repetitions=1)
        gate_metrics = result["variants"]["kylin_persisted"]
        rounds.append({
            "round": index + 1,
            "started_at_epoch": started,
            "gate_metrics": gate_metrics,
            "result": result,
        })
    payload = {
        "schema_version": "retrieval-evidence-v2",
        "git_head": git_head(),
        "datasets": {
            "seed": {"path": str(SEED_PATH), "sha256": sha256_file(SEED_PATH)},
            "query": {"path": str(QUERY_PATH), "sha256": sha256_file(QUERY_PATH)},
        },
        "rounds": rounds,
        "gate": evaluate_gate([row["gate_metrics"] for row in rounds]),
    }
    args.output.parent.mkdir(parents=True, exist_ok=True)
    args.output.write_text(json.dumps(payload, ensure_ascii=False, indent=2), encoding="utf-8")
    return 0 if payload["gate"]["pass"] else 2


if __name__ == "__main__":
```
